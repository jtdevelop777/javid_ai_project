# worker.py
import zmq
import requests

context = zmq.Context()
receiver = context.socket(zmq.PULL)
receiver.connect("tcp://localhost:5555") # เชื่อมไปที่ท่อที่ Gateway เปิดไว้

print("Javid Worker: สแตนบายรอรับงานจากกัปตันแล้ว...")

while True:
    task = receiver.recv_json()
    prompt = task['prompt']
    print(f"ได้รับงาน: {prompt}")
    
    # ยิงไปที่ Ollama API (รันด้วย GPU RX 470)
    try:
        response = requests.post(
            "http://localhost:11434/api/generate",
            json={"model": "llama3", "prompt": prompt, "stream": False},
            timeout=60
        )
        print(f"Javid ตอบ: {response.json().get('response')}")
    except Exception as e:
        print(f"Error: {e}")

Python
import zmq
import time

def start_worker():
    context = zmq.Context()
    # ใช้ REP (Reply) เพื่อรับงานจาก FastAPI
    socket = context.socket(zmq.REP)
    socket.connect("tcp://localhost:5555")
    
    print("Javid AI Worker: พร้อมรับงานแล้วกัปตัน!")
    
    while True:
        # รับข้อความจาก FastAPI
        message = socket.recv_string()
        print(f"ได้รับงาน: {message}")
        
        # ส่วนนี้เดี๋ยวเราจะเอาไปเชื่อมกับ Ollama (OpenCL)
        # ตอนนี้จำลองการทำงานไปก่อน
        time.sleep(1) 
        response = f"Javid ตอบกลับ: กำลังประมวลผล '{message}' ผ่าน OpenCL..."
        
        # ส่งคำตอบกลับไป
        socket.send_string(response)

if __name__ == "__main__":
    start_worker()

import zmq
import requests

context = zmq.Context()
# เปลี่ยนจาก PULL เป็น SUB หรือ REP ตามความเหมาะสม แต่เริ่มที่ PULL ก่อนง่ายดีครับ
receiver = context.socket(zmq.PULL)
receiver.connect("tcp://localhost:5555")

print("Javid Worker: สแตนบายรอรับงานจากกัปตันแล้ว...")

while True:
    task = receiver.recv_json()
    prompt = task.get("prompt")
    print(f"กำลังส่งงานไปที่ Docker: {prompt}")
    
    # ยิงไปที่ Ollama ใน Docker
    try:
        response = requests.post(
            "http://localhost:11434/api/generate",
            json={"model": "llama3", "prompt": prompt, "stream": False}
        )
        print(f"AI ตอบกลับมาแล้วครับกัปตัน!")
    except Exception as e:
        print(f"เกิดข้อผิดพลาด: {e}")

from fastapi import FastAPI
import zmq

app = FastAPI(title="Javid AI Gateway")

# เตรียมท่อ ZeroMQ
context = zmq.Context()
sender = context.socket(zmq.PUSH)
sender.bind("tcp://*:5555") # ผูกไว้ที่ Port 5555 เพื่อรอ Worker มารับงาน

@app.get("/ask")
async def ask_javid(prompt: str):
    # ส่งงานลงท่อ (JSON format)
    sender.send_json({"prompt": prompt})
    return {"status": "Success", "msg": f"ส่งงาน '{prompt}' เข้าคิวแล้วครับกัปตัน!"}

3. เขียน Code: AI Worker (หลังบ้าน)
สร้างไฟล์ worker.py ตัวนี้จะดึงงานจาก ZeroMQ ไปถาม Ollama ที่เราแก้เรื่อง GPU ไว้แล้วครับ

Python
# worker.py
import zmq
import requests

context = zmq.Context()
receiver = context.socket(zmq.PULL)
receiver.connect("tcp://localhost:5555") # เชื่อมไปที่ท่อที่ Gateway เปิดไว้

print("Javid Worker: สแตนบายรอรับงานจากกัปตันแล้ว...")

while True:
    task = receiver.recv_json()
    prompt = task['prompt']
    print(f"ได้รับงาน: {prompt}")
    
    # ยิงไปที่ Ollama API (รันด้วย GPU RX 470)
    try:
        response = requests.post(
            "http://localhost:11434/api/generate",
            json={"model": "llama3", "prompt": prompt, "stream": False},
            timeout=60
        )
        print(f"Javid ตอบ: {response.json().get('response')}")
    except Exception as e:
        print(f"Error: {e}")
